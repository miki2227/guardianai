 116|        // get the current config so if the file isn't found
 117|        // the first time the hash will be added to output file
 118|        configHash = getHashCodeBasedOnObjectContent(config);
 119|        final Path path = Path.of(fileName);
 120|        if (Files.exists(path)) {

 122|                details.load(inStream);
 123|                final String cachedConfigHash = details.getProperty(CONFIG_HASH_KEY);
 124|                if (!configHash.equals(cachedConfigHash)) {
 125|                    // Detected configuration change - clear cache
 126|                    reset();
 127|                }
 128|            }
 129|        }
 130|        else {
 131|            // put the hash in the file if the file is going to be created
 132|            reset();
 133|        }
 134|    }

 212|     * Calculates the hashcode for the serializable object based on its content.
 213|     *
 214|     * @param object serializable object.
 215|     * @return the hashcode for serializable object.
 216|     * @throws IllegalStateException when some unexpected happened.
 217|     */
 218|    private static String getHashCodeBasedOnObjectContent(Serializable object) {
 219|        try {
 220|            final ByteArrayOutputStream outputStream = new ByteArrayOutputStream();
 221|            // in-memory serialization of Configuration
 222|            serialize(object, outputStream);
 223|            // Instead of hexEncoding outputStream.toByteArray() directly we
 224|            // use a message digest here to keep the length of the
 225|            // hashcode reasonable
 226|
 227|            final MessageDigest digest = MessageDigest.getInstance("SHA-1");
 228|            digest.update(outputStream.toByteArray());
 229|
 230|            return new BigInteger(1, digest.digest()).toString(BASE_16).toUpperCase(Locale.ROOT);
 231|        }
 232|        catch (final IOException | NoSuchAlgorithmException exc) {
 233|            // rethrow as unchecked exception
 234|            throw new IllegalStateException("Unable to calculate hashcode for "
 235|                    + object.getClass().getName(), exc);
 236|        }
 237|    }
