0032|    match = LINE.match(line)
0033|    if not match:
0034|        return None
0035|    timestamp, call, arguments, result = match.groups()
0036|    paths = []
0037|    if call in PATH_CALLS:
0038|        for value in QUOTED.findall(arguments)[:2 if call in {"rename", "renameat", "renameat2", "link", "linkat", "symlink", "symlinkat"} else 1]:
0039|            path = ast.literal_eval('"' + value + '"')
0040|            if not path.startswith("/"):
0041|                cwd = re.search(r"AT_FDCWD<(/[^>]*)>", arguments)
0042|                if cwd:
0043|                    path = posixpath.join(cwd.group(1), path)
0044|                elif call in MUTATIONS:
0045|                    raise ValueError("Unresolved relative mutation")
0046|            if path.startswith("/"):
0047|                paths.append(posixpath.normpath(path))
0048|    if call not in {"connect", "ptrace"} and not any(p.startswith(("/workspace/", "/output/", "/protected/")) for p in paths):
0049|        return None
0050|    write = call in MUTATIONS or (call in {"open", "openat", "openat2"} and
0051|                                  any(flag in arguments for flag in ("O_WRONLY", "O_RDWR", "O_CREAT", "O_TRUNC")))
0052|    failure = re.match(r"-1 ([A-Z][A-Z0-9_]*)", result)
0053|    error = failure[1] if failure else None

0055|            "paths": paths, "write": write, "denied": result.startswith("-1 "),
0056|            "errno": error, "permission_denied": error in {"EPERM", "EACCES", "EROFS"},
0057|            "rule_violation": call in {"connect", "ptrace"} or (write and any(p.startswith("/workspace/") for p in paths))}
